cmake_minimum_required(VERSION 3.22)
project(Vita3KVulkanSurfaceTests LANGUAGES CXX)
enable_testing()
find_package(Vulkan REQUIRED)
add_executable(vulkan_surface_tests VulkanSurfaceTests.cpp)
target_compile_features(vulkan_surface_tests PRIVATE cxx_std_20)
target_compile_definitions(vulkan_surface_tests PRIVATE VULKAN_HPP_NO_CONSTRUCTORS)
target_include_directories(vulkan_surface_tests PRIVATE ../../../vita3k/renderer/include)
target_link_libraries(vulkan_surface_tests PRIVATE Vulkan::Vulkan)
add_test(NAME vulkan_surface_tests COMMAND vulkan_surface_tests)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME surface_cache_production_tests COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/SurfaceCacheProductionTests.py ${CMAKE_CXX_COMPILER})
set_tests_properties(vulkan_surface_tests surface_cache_production_tests PROPERTIES TIMEOUT 60)
