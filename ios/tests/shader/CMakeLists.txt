cmake_minimum_required(VERSION 3.22)
project(Vita3KShaderRegressionTests LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(VITA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
foreach(dep glslang SPIRV-Cross fmt spdlog boost)
    if(NOT EXISTS "${VITA_ROOT}/external/${dep}/.git")
        message(FATAL_ERROR "Initialize pinned external/${dep} submodule; see README.md")
    endif()
endforeach()
set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
set(SPDLOG_FMT_EXTERNAL ON CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_MSL ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_CPP OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_REFLECT OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_SKIP_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${VITA_ROOT}/external/fmt" fmt)
add_subdirectory("${VITA_ROOT}/external/spdlog" spdlog)
add_subdirectory("${VITA_ROOT}/external/glslang" glslang)
add_subdirectory("${VITA_ROOT}/external/SPIRV-Cross" spirv-cross)
# Use production diagnostics and types. Only the optional CLI file reader is supplied below.
add_library(util INTERFACE)
target_include_directories(util INTERFACE
    "${VITA_ROOT}/vita3k/util/include" "${VITA_ROOT}/vita3k/mem/include"
    "${VITA_ROOT}/external/boost")
target_link_libraries(util INTERFACE fmt spdlog)
add_subdirectory("${VITA_ROOT}/vita3k/features" features)
add_subdirectory("${VITA_ROOT}/vita3k/gxm" gxm)
add_subdirectory("${VITA_ROOT}/vita3k/shader" shader)
# Exercise the iOS shader paths with the host SPIR-V/MSL tools.
target_compile_definitions(shader PRIVATE VITA3K_PLATFORM_IOS=1)
add_library(shader_test_paths STATIC
    "${VITA_ROOT}/external/boost/libs/filesystem/src/path.cpp"
    "${VITA_ROOT}/external/boost/libs/filesystem/src/path_traits.cpp"
    "${VITA_ROOT}/external/boost/libs/filesystem/src/codecvt_error_category.cpp")
target_include_directories(shader_test_paths PRIVATE "${VITA_ROOT}/external/boost")
add_executable(shader_regression_tests ShaderRegressionTests.cpp FileReader.cpp)
target_include_directories(shader_regression_tests PRIVATE "${VITA_ROOT}/external/glslang")
target_link_libraries(shader_regression_tests PRIVATE shader SPIRV spirv-cross-msl shader_test_paths)
add_test(NAME shader_regression_tests COMMAND shader_regression_tests "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(shader_regression_tests PROPERTIES TIMEOUT 60)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(SPIRV_VAL spirv-val REQUIRED)
find_program(SPIRV_DIS spirv-dis REQUIRED)
set_tests_properties(shader_regression_tests PROPERTIES FIXTURES_SETUP generated_shaders)
add_test(NAME shader_spirv_validation COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/ValidateSpirv.py" "${CMAKE_CURRENT_BINARY_DIR}"
    "${SPIRV_VAL}" "${SPIRV_DIS}")
set_tests_properties(shader_spirv_validation PROPERTIES FIXTURES_REQUIRED generated_shaders TIMEOUT 60)

# Execute the production-generated byte loads when a host Vulkan driver is available.
option(VITA3K_EXECUTE_SHADER_TESTS "Execute generated uniform shaders on Vulkan" OFF)
if(VITA3K_EXECUTE_SHADER_TESTS)
    find_package(Vulkan REQUIRED)
    add_executable(uniform_execution_tests UniformExecutionTests.cpp)
    target_link_libraries(uniform_execution_tests PRIVATE Vulkan::Vulkan)
    add_test(NAME uniform_execution_tests COMMAND uniform_execution_tests "${CMAKE_CURRENT_BINARY_DIR}/unmapped_fetch.spv")
    set_tests_properties(uniform_execution_tests PROPERTIES FIXTURES_REQUIRED generated_shaders TIMEOUT 60)
endif()
