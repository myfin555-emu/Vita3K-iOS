cmake_minimum_required(VERSION 3.22)

option(VITA3K_BUILD_IOS "Build the experimental iOS bootstrap application" OFF)
option(VITA3K_BUILD_IOS_UPSTREAM_CORE "Build the upstream Vita3K core for iOS" OFF)
option(VITA3K_ENABLE_LTO "Enable link-time optimization when supported" ON)

# Set the minimum before project()/toolchain initialization so every target agrees.
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(VITA3K_IOS_DEPLOYMENT_TARGET "16.4" CACHE STRING "Minimum supported iOS version")
    if(VITA3K_IOS_DEPLOYMENT_TARGET VERSION_LESS "16.4")
        message(FATAL_ERROR "Vita3K iOS requires iOS 16.4 or later")
    endif()
    set(CMAKE_OSX_DEPLOYMENT_TARGET "${VITA3K_IOS_DEPLOYMENT_TARGET}" CACHE STRING "iOS minimum" FORCE)
endif()

if(VITA3K_BUILD_IOS_UPSTREAM_CORE)
	if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
		message(FATAL_ERROR "VITA3K_BUILD_IOS_UPSTREAM_CORE requires an iOS toolchain.")
	endif()
	set(VITA3K_PLATFORM_IOS 1)
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS" OR VITA3K_BUILD_IOS)
	if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
		message(FATAL_ERROR "VITA3K_BUILD_IOS requires an iOS toolchain. Configure with -DCMAKE_SYSTEM_NAME=iOS on macOS with Xcode installed.")
	endif()
	project(Vita3KiOS LANGUAGES C CXX OBJC OBJCXX)
	add_subdirectory(ios)
	return()
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Android")
	set(ANDROID 1)
	set(VCPKG_TARGET_ANDROID 1)
	include("cmake/vcpkg_android.cmake")
endif()

project(Vita3K)

if(VITA3K_BUILD_IOS_UPSTREAM_CORE AND NOT CMAKE_SYSTEM_PROCESSOR)
	set(CMAKE_SYSTEM_PROCESSOR arm64)
endif()

include(ProcessorCount)
ProcessorCount(CPU_COUNT)
include("external/dynarmic/CMakeModules/DetectArchitecture.cmake")

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(CMAKE_OSX_DEPLOYMENT_TARGET 13.3)
endif()

if(NOT MSVC)
    string(APPEND CMAKE_C_FLAGS_RELEASE " -O3 -DNDEBUG")
    string(APPEND CMAKE_CXX_FLAGS_RELEASE " -O3 -DNDEBUG")
    string(APPEND CMAKE_C_FLAGS_RELWITHDEBINFO " -O3 -DNDEBUG")
    string(APPEND CMAKE_CXX_FLAGS_RELWITHDEBINFO " -O3 -DNDEBUG")
    string(APPEND CMAKE_C_FLAGS_MINSIZEREL " -Os -DNDEBUG")
    string(APPEND CMAKE_CXX_FLAGS_MINSIZEREL " -Os -DNDEBUG")
endif()

if(NOT MSVC)
    string(APPEND CMAKE_C_FLAGS_RELEASE " -fvisibility=hidden")
    string(APPEND CMAKE_CXX_FLAGS_RELEASE " -fvisibility=hidden -fvisibility-inlines-hidden")
endif()

# workaround for SDL due to cmake bug
if(APPLE)
    enable_language(OBJC)
    if(VITA3K_BUILD_IOS_UPSTREAM_CORE)
        enable_language(OBJCXX)
        enable_language(Swift)
        if(CMAKE_Swift_COMPILER_VERSION VERSION_LESS 5.9)
            message(FATAL_ERROR
                "The iOS frontend needs Swift 5.9 or newer (found ${CMAKE_Swift_COMPILER_VERSION}.)")
        endif()
    endif()
endif()

if(VITA3K_BUILD_IOS_UPSTREAM_CORE)
	add_compile_definitions(VITA3K_PLATFORM_IOS=1)
	set(CMAKE_XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC "YES")
endif()

if(ANDROID)
	set(CMAKE_SYSROOT "")
	set(CMAKE_OSX_SYSROOT "")
	set_property(GLOBAL PROPERTY NO_SYSTEM_FROM_IMPORTED ON)
	string(APPEND CMAKE_CXX_FLAGS " -Wl,-Bsymbolic -Wno-unused-command-line-argument")
	string(APPEND CMAKE_C_FLAGS " -Wl,-Bsymbolic -Wno-unused-command-line-argument")
	if(CMAKE_BUILD_TYPE STREQUAL "Release")
		set(CMAKE_C_VISIBILITY_PRESET hidden)
		set(CMAKE_CXX_VISIBILITY_PRESET hidden)
		set(CMAKE_VISIBILITY_INLINES_HIDDEN 1)
	endif()
else()
	if((ARCHITECTURE STREQUAL "x86_64") OR APPLE)
		option(USE_DISCORD_RICH_PRESENCE "Build Vita3K with Discord Rich Presence" ON)
	endif()
	option(BUILD_APPIMAGE "Build an AppImage." OFF)
endif()

option(FORCE_BUILD_OPENSSL_MAC OFF)

if("${CMAKE_CXX_COMPILER_LAUNCHER}" STREQUAL "")
	find_program(CCACHE_PROGRAM ccache)
	if(CCACHE_PROGRAM)
		set(CMAKE_C_COMPILER_LAUNCHER "${CCACHE_PROGRAM}")
		set(CMAKE_CXX_COMPILER_LAUNCHER "${CCACHE_PROGRAM}")
	endif()
endif()

# Keep the remainder of the upstream CMake configuration unchanged.
